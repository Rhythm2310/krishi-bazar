# Paste into Jules: repo page > Configuration > Initial Setup, then click "Run and Snapshot".
# Idempotent and fast. Frozen installs only (versions pinned in .jules/deps.csv). No secrets in this file.
set -euo pipefail

node --version     # Jules preinstalls Node; deps.csv pins the 24 line (CI runs 24.22.1)
python3 --version  # the .jules/tools guards need Python >= 3.9 (deps.csv pins 3.12)

if [ -f package.json ]; then
  npm ci           # frozen install from the committed lockfile
else
  echo "No package.json yet (pre-foundation commit) - nothing to install."
fi

if [ -f playwright.config.ts ]; then
  npx playwright install --with-deps chromium || echo "Playwright browser install failed here; run 'npx playwright install --with-deps chromium' inside the task session before the e2e tests."
fi
